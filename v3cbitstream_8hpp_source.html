23static inline int floorLog2(uint32_t x) {
24 if (x == 0) {
25 return -1;
26 }
27#ifdef __GNUC__
28 return 31 - __builtin_clz(x);
29#else
30 int result = 0;
31 if (x & 0xffff0000) {
32 x >>= 16;
33 result += 16;
34 }
35 if (x & 0xff00) {
36 x >>= 8;
37 result += 8;
38 }
39 if (x & 0xf0) {
40 x >>= 4;
41 result += 4;
42 }
43 if (x & 0xc) {
44 x >>= 2;
45 result += 2;
46 }
47 if (x & 0x2) {
48 result += 1;
49 }
50 return result;
51#endif
52}
